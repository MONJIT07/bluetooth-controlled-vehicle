/*
  Arduino Bluetooth Controlled Car
  L298N Motor Driver + HC-05 Bluetooth Module

  Commands:
  F = Forward
  B = Backward
  L = Left
  R = Right
  S = Stop
*/

#include <SoftwareSerial.h>

// ---------------- BLUETOOTH ----------------
// Arduino RX = 9
// Arduino TX = 10
SoftwareSerial bluetoothSerial(9, 10); // RX, TX

// ---------------- L298N MOTOR DRIVER ----------------

// Left-side motors
#define ENA 5
#define IN1 6
#define IN2 7

// Right-side motors
#define ENB 3
#define IN3 8
#define IN4 4

// Motor speed: 0 to 255
int motorSpeed = 255;

char command;


// ---------------- SETUP ----------------
void setup()
{
  // L298N pins
  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(ENB, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  // Bluetooth communication
  bluetoothSerial.begin(9600);

  // Start with motors stopped
  Stop();
}


// ---------------- MAIN LOOP ----------------
void loop()
{
  if (bluetoothSerial.available() > 0)
  {
    command = bluetoothSerial.read();

    switch (command)
    {
      case 'F':
        forward();
        break;

      case 'B':
        back();
        break;

      case 'L':
        left();
        break;

      case 'R':
        right();
        break;

      case 'S':
        Stop();
        break;
    }
  }
}


// ---------------- FORWARD ----------------
void forward()
{
  // Set motor speed
  analogWrite(ENA, motorSpeed);
  analogWrite(ENB, motorSpeed);

  // Left motor forward
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  // Right motor forward
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}


// ---------------- BACKWARD ----------------
void back()
{
  // Set motor speed
  analogWrite(ENA, motorSpeed);
  analogWrite(ENB, motorSpeed);

  // Left motor backward
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  // Right motor backward
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}


// ---------------- LEFT ----------------
void left()
{
  analogWrite(ENA, motorSpeed);
  analogWrite(ENB, motorSpeed);

  // Left side backward
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  // Right side forward
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}


// ---------------- RIGHT ----------------
void right()
{
  analogWrite(ENA, motorSpeed);
  analogWrite(ENB, motorSpeed);

  // Left side forward
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  // Right side backward
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}


// ---------------- STOP ----------------
void Stop()
{
  // Stop both motors
  analogWrite(ENA, 0);
  analogWrite(ENB, 0);

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}
